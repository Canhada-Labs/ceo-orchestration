+"""Additive, bounded shell-command position scanner for PLAN-195.
+
+This is deliberately not a shell evaluator: it follows only source-visible
+command positions and shell bodies, and never expands variables or executes
+subprocesses.  Unrecognised input is left to the existing guards.
+"""
+
+from __future__ import annotations
+
+import os
+import fnmatch
+import itertools
+import re
+import shlex
+from dataclasses import dataclass
+from typing import List, Optional
+
+
+MAX_DEPTH = 8
+MAX_BODY_BYTES = 16384
+MAX_WORDS = 1024
+SEPARATORS = frozenset((";", "|", "||", "&", "&&", "(", ")", "{", "}", "\n"))
+SHELLS = frozenset(("sh", "bash", "dash", "zsh", "ksh"))
+RESERVED = frozenset((
+    "if", "then", "elif", "else", "fi", "while", "until", "do", "done",
+    "for", "in", "case", "esac", "select", "time", "!", "function",
+))
+ASSIGNMENT = re.compile(r"^[A-Za-z_][A-Za-z_0-9]*=")
+HEREDOC = re.compile(r"<<-?\s*(['\"]?)([A-Za-z_][A-Za-z_0-9]*)\1[^\n]*\n")
+STATIC_PRINTF = re.compile(
+    r"\$\(\s*printf\s+(?:['\"]%s['\"]\s+)?([A-Za-z0-9_./-]+)\s*\)"
+)
+VARIABLE = re.compile(r"\$(?:\{([A-Za-z_][A-Za-z_0-9]*)\}|([A-Za-z_][A-Za-z_0-9]*))")
+_QUOTED_OPERATORS = {ch: chr(0xF100 + index) for index, ch in enumerate(";&|(){}")}
+_RESTORE_OPERATORS = {value: key for key, value in _QUOTED_OPERATORS.items()}
+
+
+@dataclass(frozen=True)
+class GuardHit:
+    reason_code: str
+    reason: str
+
+
+def _hit(code: str) -> GuardHit:
+    return GuardHit(
+        code,
+        "BLOCKED: indirect shell execution (%s). Write the executable "
+        "command in direct form so the Bash safety guard can judge it."
+        % code,
+    )
+

+def _prelex(command: str) -> str:
+    """Keep literal mid-word # and newline boundaries; strip true comments.
+
+    The shell opens a comment only outside quotes at the start of a word.
+    An escaped # never introduces a comment.  Quote characters occurring
+    inside comments do not change parser state.
+    """
+    out: List[str] = []
+    quote = ""
+    escaped = False
+    word_start = True
+    commenting = False
+    for ch in command:
+        if commenting:
+            if ch == "\n":
+                out.append(ch)
+                commenting = False
+                word_start = True
+            continue
+        if escaped:
+            out.append(ch)
+            escaped = False
+            word_start = False
+            continue
+        if ch == "\\" and quote != "'":
+            out.append(ch)
+            escaped = True
+            continue
+        if ch == quote:
+            quote = ""
+            out.append(ch)
+            word_start = False
+            continue
+        if ch in ("'", '"') and not quote:
+            quote = ch
+            out.append(ch)
+            word_start = False
+            continue
+        if ch == "#" and not quote and word_start:
+            commenting = True
+            continue
+        out.append(ch)
+        if not quote:
+            word_start = ch.isspace() or ch in ";&|(){}<>"
+    return "".join(out)
+
+
+def _words(command: str) -> List[str]:
+    text = _prelex(command)
+    shielded: List[str] = []
+    quote = ""
+    escaped = False
+    for ch in text:
+        if escaped:
+            shielded.append(ch)
+            escaped = False
+        elif ch == "\\" and quote != "'":
+            shielded.append(ch)
+            escaped = True
+        elif ch == quote:
+            quote = ""
+            shielded.append(ch)
+        elif ch in ("'", '"') and not quote:
+            quote = ch
+            shielded.append(ch)
+        else:
+            shielded.append(_QUOTED_OPERATORS.get(ch, ch) if quote else ch)
+    lexer = shlex.shlex("".join(shielded), posix=True,
+                        punctuation_chars=";&|()<>\n")
+    lexer.whitespace = " \t\r"
+    lexer.whitespace_split = True
+    lexer.commenters = ""
+    words = list(lexer)
+    if len(words) > MAX_WORDS:
+        raise ValueError("word ceiling")
+    return words
+
+
+def _unshield(value: str) -> str:
+    for hidden, literal in _RESTORE_OPERATORS.items():
+        value = value.replace(hidden, literal)
+    return value
+

+def _command_substitutions(command: str, depth: int,
+                           heuristics: bool = True,
+                           source_comments: bool = True) -> Optional[GuardHit]:
+    """Inspect executable $(...) even inside double quotes, never single."""
+    # Strip comments in shell SOURCE.  Unquoted here-doc bodies are DATA:
+    # their `# $(...)` still undergoes command substitution before delivery.
+    if source_comments:
+        command = _prelex(command)
+    index = 0
+    quote = ""
+    while index < len(command):
+        ch = command[index]
+        if ch == "\\" and quote != "'":
+            index += 2
+            continue
+        if ch == quote:
+            quote = ""
+        elif ch in ("'", '"') and not quote:
+            quote = ch
+        elif (quote != "'" and command[index:index + 2] == "$(") or (
+            not quote and command[index:index + 2] in ("<(", ">(")
+        ):
+            start = index + 2
+            level = 1
+            cursor = start
+            while cursor < len(command):
+                if command[cursor:cursor + 2] == "$(":
+                    level += 1
+                    cursor += 2
+                    continue
+                if command[cursor] == ")":
+                    level -= 1
+                    if level == 0:
+                        hit = _scan(command[start:cursor], depth + 1,
+                                    strict=True, heuristics=heuristics)
+                        if hit:
+                            return hit
+                        index = cursor + 1
+                        break
+                cursor += 1
+            else:
+                return _hit("bash_indirect_unparseable_shell")
+            continue
+        elif ch == "`" and quote != "'":
+            cursor = index + 1
+            while cursor < len(command):
+                if command[cursor] == "\\":
+                    cursor += 2
+                    continue
+                if command[cursor] == "`":
+                    hit = _scan(command[index + 1:cursor], depth + 1,
+                                strict=True, heuristics=heuristics)
+                    if hit:
+                        return hit
+                    index = cursor + 1
+                    break
+                cursor += 1
+            else:
+                return _hit("bash_indirect_unparseable_shell")
+            continue
+        index += 1
+    return None

+def _extract_heredocs(command: str, depth: int,
+                      heuristics: bool = True) -> tuple:
+    """Strip here-doc DATA; parse a shell consumer's PROGRAM body.
+
+    A nonquoted delimiter expands command substitutions even when the
+    consumer is a data-only tool.  This is separate from interpreting the
+    heredoc as shell source, which only a visible shell consumer does.
+    """
+    bodies: List[str] = []
+    output: List[str] = []
+    cursor = 0
+    while True:
+        match = HEREDOC.search(command, cursor)
+        if match is None:
+            output.append(command[cursor:])
+            break
+        begin = match.end()
+        closing = re.search(r"(?m)^" + re.escape(match.group(2)) + r"[ \t]*(?:\n|$)",
+                            command[begin:])
+        if closing is None:
+            output.append(command[cursor:])
+            break
+        body = command[begin:begin + closing.start()]
+        prefix = command[cursor:match.start()]
+        output.append(prefix + command[match.start():begin])
+        consumer_tokens = prefix.strip().split()
+        consumer = os.path.basename(consumer_tokens[0]) if consumer_tokens else ""
+        if not match.group(1):
+            substitutions = _command_substitutions(
+                body, depth, heuristics, source_comments=False
+            )
+            if substitutions:
+                return "", [substitutions]
+        if consumer in SHELLS or consumer == "eval":
+            bodies.append(body)
+        cursor = begin + closing.end()
+    return "".join(output), bodies

+def _normalise(argv: List[str]) -> List[str]:
+    """Consume known launchers, flags, and same-command env assignments."""
+    tokens = list(argv)
+    for _ in range(12):
+        if not tokens:
+            break
+        head = os.path.basename(tokens[0].lstrip("\\")).lower()
+        if ASSIGNMENT.match(tokens[0]):
+            tokens.pop(0)
+            continue
+        if head in ("sudo", "doas", "nocorrect", "command", "builtin", "nohup"):
+            tokens.pop(0)
+            while tokens and tokens[0].startswith("-"):
+                flag = tokens.pop(0)
+                if flag in ("-u", "--user", "-g", "--group") and tokens:
+                    tokens.pop(0)
+            continue
+        if head == "env":
+            tokens.pop(0)
+            while tokens and (tokens[0].startswith("-") or ASSIGNMENT.match(tokens[0])):
+                flag = tokens.pop(0)
+                if flag in ("-u", "--unset", "-C", "--chdir") and tokens:
+                    tokens.pop(0)
+            continue
+        tokens[0] = head
+        break
+    return tokens

+def _shell_body(argv: List[str]) -> Optional[str]:
+    if not argv or argv[0] not in SHELLS:
+        return None

+def _brace_commands(token: str) -> List[str]:
+    opening = token.find("{")
+    closing = token.find("}", opening + 1)
+    if opening < 0 or closing < 0:
+        return [token]
+    body = token[opening + 1:closing]
+    if "," in body:
+        variants = body.split(",")
+    else:
+        numeric = re.fullmatch(r"(-?\d+)\.\.(-?\d+)(?:\.\.(-?\d+))?", body)
+        if numeric is None:
+            return [token]
+        first, last = int(numeric.group(1)), int(numeric.group(2))
+        # Bash takes the magnitude of an explicit step, chooses direction
+        # from the endpoints, and treats a zero step as one.
+        stride = abs(int(numeric.group(3))) if numeric.group(3) else 1
+        step = (stride or 1) * (1 if first <= last else -1)
+        count = abs(last - first) // abs(step) + 1
+        if count > 128:
+            raise ValueError("brace ceiling")
+        variants = [str(first + i * step) for i in range(count)]
+    if len(variants) > 128:
+        raise ValueError("brace ceiling")
+    return [token[:opening] + item + token[closing + 1:] for item in variants]

+def _scan_segment(argv: List[str], depth: int,
+                  heuristics: bool = True) -> Optional[GuardHit]:
+    if not argv:
+        return None

+                return _hit("bash_indirect_ambiguous_launcher")

+    except ValueError:

+                tail = normal[index + 1:]

+                                     depth + 1, heuristics)

+        w in ("--hard", "--force", "-rf", "-fr", "-r", "-f") for w in normal[1:]

+def _canonical_target(target: str, cwd: str, assignments: dict) -> bool:
+    """Bounded textual resolution: no glob expansion or filesystem access."""
+    try:
+        from check_canonical_edit import _CANONICAL_GUARDS, _fnmatch_segments
+    except ImportError:
+        return False  # infrastructure failures must fail-open
+    value = target.strip("'\"")
+    value = VARIABLE.sub(
+        lambda match: assignments.get(match.group(1) or match.group(2), ""), value
+    )
+    if not value:
+        return False
+    if value.startswith("/"):
+        project = os.environ.get("CLAUDE_PROJECT_DIR", "")
+        if not project or not value.startswith(project.rstrip("/") + "/"):
+            return False
+        value = value[len(project.rstrip("/")) + 1:]
+    else:
+        value = os.path.normpath(os.path.join(cwd, value))
+    for pattern in _CANONICAL_GUARDS:
+        if _fnmatch_segments(value, pattern):
+            return True

+        # characters to avoid locking out ordinary variable-driven scripts.
+        if any(ch in value for ch in "*?[") and sum(
+            ch.isalnum() for ch in value
+        ) >= 4 and not any(ch in pattern for ch in "*?["):
+            if fnmatch.fnmatchcase(pattern.lower(), value.lower()):
+                return True
+    return False

+def _computed_target(argv: List[str], cwd: str, assignments: dict) -> Optional[GuardHit]:
+    if not argv:
+        return None
+    targets: List[str] = []
+    for idx, word in enumerate(argv[:-1]):
+        if word in (">", ">>", ">|", "&>", "<>" ):

+    if normal and normal[0] == "sed" and any(
+        arg.startswith("-i") for arg in normal[1:]

+    if normal and normal[0] == "dd":
+        targets += [arg[3:] for arg in normal[1:] if arg.startswith("of=")]
+    if normal and normal[0] in ("python", "python3", "node", "ruby", "perl"):
+        for index, argument in enumerate(normal[1:], 1):
+            if argument in ("-c", "-e", "-E", "--eval") and index + 2 < len(normal):
+                for operand in normal[index + 2:]:
+                    if _canonical_target(operand, cwd, assignments):
+                        return _hit("bash_computed_canonical_target")
+                break
+    for target in targets:
+        if _canonical_target(target, cwd, assignments) and (
+            "$" in target or any(ch in target for ch in "*?[") or cwd
+        ):
+            return _hit("bash_computed_canonical_target")
+    return None

+def _scan(command: str, depth: int, *, strict: bool = False,
+          heuristics: bool = True) -> Optional[GuardHit]:
+    if depth > MAX_DEPTH or len(command) > MAX_BODY_BYTES:
+        return _hit("bash_indirect_limit")
+    command, heredoc_bodies = _extract_heredocs(command, depth, heuristics)
+    for body in heredoc_bodies:
+        if isinstance(body, GuardHit):
+            return body
+        hit = _scan(body, depth + 1, strict=True, heuristics=heuristics)
+        if hit:
+            return hit
+    substitution_hit = _command_substitutions(command, depth, heuristics)
+    if substitution_hit:
+        return substitution_hit
+    # A small, statically decidable output producer for same-command targets.
+    # No process is started and anything beyond this exact printf form stays
+    # unresolved (and therefore is not blindly blocked).
+    command = STATIC_PRINTF.sub(lambda match: match.group(1), command)
+    try:
+        words = _words(command)
+    except ValueError as exc:
+        if str(exc) == "word ceiling":
+            return _hit("bash_indirect_limit")
+        return _hit("bash_indirect_unparseable_shell") if strict else None
+    segment: List[str] = []
+    cwd = ""
+    assignments: dict = {}
+    aliases: dict = {}
+    previous_separator = ""
+    for token in words + [";"]:
+        if token in SEPARATORS or (token and all(ch in ";&|(){}" for ch in token)):
+            normal = _normalise(segment)
+            if normal and normal[0] == "alias":
+                for definition in normal[1:]:
+                    if ASSIGNMENT.match(definition):
+                        name, value = definition.split("=", 1)
+                        aliases[name] = _unshield(value)
+            elif normal and normal[0] in aliases:
+                expanded_alias = aliases[normal[0]] + " " + " ".join(
+                    shlex.quote(_unshield(arg)) for arg in normal[1:]
+                )
+                hit = _scan(expanded_alias, depth + 1,
+                            strict=True, heuristics=heuristics)
+                if hit:
+                    return hit
+            if normal and normal[0] in SHELLS and _shell_body(normal) is None:
+                if "<<<" in segment:
+                    position = segment.index("<<<")
+                    if position + 1 < len(segment):
+                        hit = _scan(_unshield(segment[position + 1]), depth + 1,
+                                    strict=True, heuristics=heuristics)
+                        if hit:
+                            return hit
+                if heuristics and (previous_separator in ("|", "|&") or "<" in segment):
+                    return _hit("bash_indirect_opaque_stdin")
+            hit = _scan_segment(segment, depth, heuristics)
+            if hit:
+                return hit
+            if segment and segment[0] == "cd" and len(segment) >= 2:
+                cwd = os.path.normpath(os.path.join(cwd, segment[1]))
+            for word in segment:
+                if ASSIGNMENT.match(word):
+                    key, value = word.split("=", 1)
+                    if "$" not in value and "`" not in value:
+                        assignments[key] = value
+            target_hit = _computed_target(segment, cwd, assignments)
+            if target_hit:
+                return target_hit
+            segment = []
+            previous_separator = token
+        elif token.lower() in RESERVED and not segment:
+            continue
+        else:
+            segment.append(token)
+    return None
+

+def scan(command: str, *, heuristics: bool = True) -> Optional[GuardHit]:
+    """Return a monotonic additional veto, never an ALLOW override."""
+    return _scan(command, 0, heuristics=heuristics)

+def _normalize_candidate_value(value: str) -> str:
+    """PLAN162_FIX_7 (finding #7) — turn a ``file://`` URI into a path.
+
+    ``uri`` is one of ``_MCP_WRITE_PATH_KEYS``, but HEAD handed the value
+    to ``Path()`` verbatim: ``file:///repo/.claude/settings.json`` became
+    a RELATIVE path whose first segment is ``file:``, resolved against the
+    process CWD, landed outside the repo root and classified
+    NON-canonical — so the edit sailed through ungated even though the
+    MCP server would write the guarded file.
+
+    Normalizing in ONE place (the single function that builds candidates)
+    was the form both debate lanes accepted; the alternative — treating an
+    un-interpretable value as fail-CLOSED — is left to the callers, which
+    already fail-CLOSE on a candidate they cannot classify.
+
+    Only LOCAL file URIs are rewritten (empty or ``localhost`` authority).
+    A ``file://remote-host/...`` URI is NOT a local path and is returned
+    untouched, as is any value with a different scheme. A percent-escape
+    that decodes to an embedded NUL is likewise returned untouched — a
+    path the OS cannot open is not made more gate-able by decoding it.
+    """
+    if len(value) < 7 or value[:7].lower() != "file://":
+        return value
+    try:
+        from urllib.parse import unquote, urlsplit  # noqa: PLC0415
+        parts = urlsplit(value)
+        if parts.scheme.lower() != "file":
+            return value
+        if (parts.netloc or "").lower() not in ("", "localhost"):
+            return value  # remote authority: not a local filesystem path
+        decoded = unquote(parts.path)
+        if not decoded or "\x00" in decoded:
+            return value
+        return decoded
+    except Exception:
+        return value

+def _repo_rels(path_str: str, repo_root: Path) -> List[str]:

+def _fnmatch_segments(path: str, pattern: str) -> bool:
+    """Segment-wise glob matcher.
+
+    - ``*`` matches exactly one path segment (any non-slash content).
+    - ``**`` matches zero or more path segments (Sprint 9 amendment).
+    - Literal segments must match exactly.
+    """
+    p_parts = path.split("/")
+    pat_parts = pattern.split("/")
+    return _match_segments(p_parts, pat_parts)
+

+def _match_segments(p_parts: List[str], pat_parts: List[str]) -> bool:
+    """Recursive glob with ``**`` zero-or-more support.
+
+    Per-segment patterns support full fnmatch semantics (so
+    patterns like *.py, *.yaml, or ADR-*.md match one
+    segment with a wildcard stem). ** still means zero-or-more
+    segments.
+    """
+    if not pat_parts:
+        return not p_parts
+    head, rest = pat_parts[0], pat_parts[1:]
+    if head == "**":
+        # Zero-or-more: try consuming 0..len(p_parts) segments
+        for i in range(len(p_parts) + 1):
+            if _match_segments(p_parts[i:], rest):
+                return True
+        return False
+    if not p_parts:
+        return False
+    # Bare-* is equivalent to fnmatch "*", but we keep the explicit
+    # branch for readability. fnmatch.fnmatchcase does case-sensitive
+    # glob on a single segment (no "/" traversal).
+    if head == "*" or fnmatch.fnmatchcase(p_parts[0], head):
+        return _match_segments(p_parts[1:], rest)
+    return False

+def _parse_scope_paths_from_text(scope_text: str) -> "Set[str]":
+    """Extract declared canonical paths from a Scope block.
+

+    region) and Tier-2 (legacy `_SCOPE_HEADER_RE`) parser paths.
+

+    top-level continuation header (Effective:, Plans:, Rationale,
+    Authorization source:, Anchor commit:, a re-encountered
+    `Approved-By:`) or markdown horizontal rule (---, ***, ___) or
+    end-of-text. Sub-headers within Scope (lines ending with `:` that

+        return None
+    if proc.returncode != 0:

+    except Exception:

+        return None
+    _ANCHOR_BLOB_CACHE[cache_key] = text

+def _pinned_sentinel_digests(src_env) -> Tuple[Set[str], bool]:
+    """``(digests, malformed)`` from ``CEO_SENTINEL_UNLOCK_SHA256``.
+
+    Empty + not-malformed means "the Owner pinned nothing" (rail B
+    decides). ANY unparseable token makes the whole value malformed:
+    a typo'd trust input is never silently narrowed to the tokens that
+    happened to parse.
+    """
+    raw = (src_env.get(_UNLOCK_DIGEST_ENV) or "").strip()
+    if not raw:
+        return (set(), False)
+    tokens = [t for t in re.split(r"[,\s]+", raw) if t]
+    digests = set()  # type: Set[str]
+    for token in tokens:
+        if not _SHA256_RE.match(token):
+            return (set(), True)
+        digests.add(token.lower())
+    return (digests, not digests)

+def _unlock_trusted_text(sentinel_path: Path, env=None) -> Tuple[bool, Optional[str]]:
+    """``(authorized, text_to_parse)`` for ONE sentinel under an unlock.
+
+    PLAN162_FIX_S2 / PLAN162_FIX_S2R2. Consulted ONLY on the
+    ``env_override`` path, where the GPG rail is skipped entirely and the
+    only thing separating an Owner-signed sentinel from one the AGENT
+    just wrote is provenance.
+
+    ``text_to_parse`` is ``None`` for "parse what is on disk" — returned
+    only by the two paths where disk bytes ARE the authorized bytes: an
+    Owner-pinned digest (which covers exactly those bytes) and the
+    non-git residual.
+    """
+    src_env = env if env is not None else os.environ
+    repo_root = _repo_root_for_sentinels()
+
+    # ---- Rail A: the Owner pinned this sentinel's CONTENT (P1-3) ----
+    pinned, malformed = _pinned_sentinel_digests(src_env)
+    if malformed:
+        _note_unlock_trust_diag("digest_malformed")
+        return (False, None)
+    if pinned:
+        try:
+            raw = sentinel_path.read_bytes()
+        except OSError:
+            _note_unlock_trust_diag("sentinel_unreadable")
+            return (False, None)
+        if hashlib.sha256(raw).hexdigest() in pinned:
+            try:
+                return (True, raw.decode("utf-8"))
+            except Exception:
+                _note_unlock_trust_diag("sentinel_undecodable")
+                return (False, None)
+        # A pin that does not cover THIS sentinel is not a verdict on the
+        # others: fall through to the anchor rail, or the pack's
+        # pre-existing sentinels would be self-DoS'd by pinning one file.
+        _note_unlock_trust_diag("digest_mismatch")
+
+    # ---- Rail B: the sentinel predates the session (anchored bytes) ----
+    try:
+        is_git = (repo_root / ".git").exists()
+    except OSError:
+        is_git = False
+    if not is_git:
+        return (True, None)  # documented residual: no anchor mechanism
+    anchor = (src_env.get(_SESSION_ANCHOR_ENV) or "").strip()
+    if not _ANCHOR_SHA_RE.match(anchor):
+        _note_unlock_trust_diag("anchor_missing" if not anchor else "anchor_malformed")
+        return (False, None)
+    try:
+        rel = str(sentinel_path.relative_to(repo_root)).replace(os.sep, "/")
+    except ValueError:
+        # A sentinel outside the repo root can never be anchored.
+        _note_unlock_trust_diag("sentinel_outside_repo")
+        return (False, None)
+    text = _anchored_blob_text(repo_root, anchor, rel)
+    if text is None:
+        _note_unlock_trust_diag("not_anchored")
+        return (False, None)
+    # P1-1: the ANCHORED bytes are the authorization, not the bytes an
+    # in-window writer left at that path.
+    return (True, text)

+    """Match ``*`` within one segment and ``**`` across zero or more."""

+            path[1:], pattern

+        path[1:], tail

+_SEMVER_RE = re.compile(r"^v?([0-9]+)\.([0-9]+)\.([0-9]+)$")
+
+
+def parse_semver(value: object) -> Optional[Tuple[int, int, int]]:
+    """Parse a stable ``major.minor.patch`` version; reject prereleases."""
+    if not isinstance(value, str):
+        return None
+    match = _SEMVER_RE.fullmatch(value.strip())
+    if match is None:
+        return None
+    return tuple(int(part) for part in match.groups())
+
+
+def compare_semver(candidate: object, floor: object) -> Optional[int]:
+    """Compare stable versions: positive when candidate is newer, else None."""
+    parsed_candidate = parse_semver(candidate)
+    parsed_floor = parse_semver(floor)
+    if parsed_candidate is None or parsed_floor is None:
+        return None
+    return (parsed_candidate > parsed_floor) - (parsed_candidate < parsed_floor)
+
+
+def compatible_package_version(
+    payload: Path, launcher: Optional[Path], manifest_version: object = None,
+    floor: str = CODEX_VERSION_FLOOR,
+) -> Optional[str]:
+    """Return a qualified stable version meeting floor and optional manifest.
+
+    This is currency evidence only: callers must not describe it as a hash
+    attestation of the executable bytes.
+    """
+    for package_file in _package_files((payload, launcher)):
+        version = _qualified_package_version(package_file)
+        comparison = compare_semver(version, floor)
+        manifest_comparison = (
+            1 if manifest_version is None else compare_semver(version, manifest_version)
+        )
+        if comparison is not None and comparison >= 0 and manifest_comparison is not None and manifest_comparison > 0:
+            return version
+    return None
+
+
+def qualified_package_version(payload: Path, launcher: Optional[Path]) -> Optional[str]:
+    """Return stable version from a nearby qualified package, if present."""
+    for package_file in _package_files((payload, launcher)):
+        version = _qualified_package_version(package_file)
+        if version is not None:
+            return version
+    return None
+
+
+def _package_files(paths: Iterable[Optional[Path]]) -> Iterable[Path]:
+    seen = set()
+    for path in paths:
+        if path is None:
+            continue
+        try:
+            path = path.resolve()
+        except OSError:
+            pass
+        for parent in (path.parent, *path.parents):
+            package_file = parent / "package.json"
+            if package_file not in seen:
+                seen.add(package_file)
+                yield package_file
+
+
+def _qualified_package_version(package_file: Path) -> Optional[str]:

+        data = json.loads(package_file.read_text(encoding="utf-8"))
+    except (OSError, ValueError, RecursionError):
+        return None

+        return None
+    version = data.get("version")
+    return version if parse_semver(version) is not None else None
